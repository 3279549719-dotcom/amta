**Goal:** 在 recall_gt.json 的 101 个 GT 框（《单翼停留之地》前 10 页）上，公平对比现役 For-Manga 与 PaddleOCR-VL-1.6 的 OCR 能力，并以 qwen-vl-ocr 作云端第三候选，输出修正常规后的 CER/EM 对比报告。

**Tech Stack:** Python（PIL/json/requests）、llama-server（llama.cpp b10582，OpenAI 兼容 :8118）、DashScope qwen-vl-ocr API、`src/amta/metrics.py`。

```python
"""在 recall_gt.json 的 101 个 GT 框上跑指定 OCR 引擎。
引擎: local(本地 llama-server :8118) | dashscope(qwen-vl-ocr)
用法: python scripts/ocr_run.py --engine local|dashscope [--pages N]
"""
from __future__ import annotations
import argparse, json, sys
from pathlib import Path
from PIL import Image

ROOT = Path(__file__).resolve().parent.parent
DATA = ROOT / "output" / "data"


def crop_regions(gt, page_paths, crop_dir):
    """从原图按 GT bbox 裁框。page_paths: {page_num(int): img_path}。返回 (crop_paths, meta)。"""
    crop_dir = Path(crop_dir); crop_dir.mkdir(parents=True, exist_ok=True)
    crops, meta = [], []
    for gkey, regions in gt["pages"].items():
        page_num = int(gkey.split("_")[1])
        src = page_paths.get(page_num)
        if src is None or not Path(src).exists():
            continue
        im = Image.open(src)
        for i, region in enumerate(regions):
            x0, y0, x1, y1 = (int(v) for v in region["bbox"])
            box = (max(0, x0 - 4), max(0, y0 - 4), min(im.width, x1 + 4), min(im.height, y1 + 4))
            fname = f"{gkey}_gt{i:02d}.png"
            crop = im.crop(box)
            p = crop_dir / fname
            crop.save(p)
            crops.append(str(p))
            meta.append({"page": page_num, "crop": str(p), "bbox": region["bbox"],
                         "content": region["content"], "type": region["type"]})
    return crops, meta
```

```python
import os, base64

def dashscope_ocr_batch(crop_paths, model="qwen-vl-ocr-latest", concurrency=4):
    """DashScope qwen-vl-ocr：OpenAI 兼容，base64 图，纯文本输出。101 张约 ¥0.1、1-3 分钟。"""
    key = os.environ["DASHSCOPE_API_KEY"]
    url = "https://dashscope.aliyuncs.com/compatible-mode/v1/chat/completions"
    out = []
    for p in crop_paths:
        with open(p, "rb") as f:
            b64 = base64.b64encode(f.read()).decode()
        payload = {
            "model": model,
            "messages": [{"role": "user", "content": [
                {"type": "image_url", "image_url": {"url": f"data:image/png;base64,{b64}"}},
                {"type": "text", "text": "OCR"},
            ]}],
        }
        r = requests.post(url, headers={"Authorization": f"Bearer {key}"}, json=payload, timeout=60)
        r.raise_for_status()
        text = r.json()["choices"][0]["message"]["content"]
        out.append({"crop": p, "ocr": text})
    return out
```

```python
"""读 pred + recall_gt meta → metrics 算 CER/EM，按 4 类+ALL 聚合 → benchmark_b_{engine}.json。"""
from __future__ import annotations
import argparse, json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parent.parent / "src"))
from amta.metrics import cer

def eval_rows(meta, preds):
    by_crop = {p["crop"]: p.get("ocr") or "" for p in preds}
    rows, stats = [], {}
    for m in meta:
        pred = by_crop.get(m["crop"], "")
        c = cer(m["content"], pred)
        em = 1 if c == 0.0 else 0
        rows.append({"crop": m["crop"], "type": m["type"], "gt": m["content"], "pred": pred, "cer": c, "em": em})
    for t in ["dialogue_in", "dialogue_out", "sfx", "bg_text"]:
        sub = [r for r in rows if r["type"] == t]
        n = len(sub)
        stats[t] = {"n": n, "cer": sum(r["cer"] for r in sub) / n if n else 0.0,
                    "em": sum(r["em"] for r in sub) / n if n else 0.0}
    n = len(rows)
    stats["ALL"] = {"n": n, "cer": sum(r["cer"] for r in rows) / n if n else 0.0,
                    "em": sum(r["em"] for r in rows) / n if n else 0.0}
    return rows, stats

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--engine", required=True)
    ap.add_argument("--preds", required=True)
    ap.add_argument("--meta", required=True)
    ap.add_argument("--out", required=True)
    a = ap.parse_args()
    preds = json.loads(Path(a.preds).read_text(encoding="utf-8"))
    meta = json.loads(Path(a.meta).read_text(encoding="utf-8"))
    rows, stats = eval_rows(meta, preds)
    json.dump({"engine": a.engine, "summary": stats, "rows": rows},
              Path(a.out).open("w", encoding="utf-8"), ensure_ascii=False, indent=2)
    print(json.dumps(stats, ensure_ascii=False, indent=2))
```
