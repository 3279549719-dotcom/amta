# Touhou Manga Translator GUI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a local web GUI for AMTA manga translation — drop images, pick Default/VLM mode, click Translate, see result, export PNG/PSD. No command line needed.

**Architecture:** FastAPI backend wraps the existing `run_pipeline.py` as a background task, serves artifacts via HTTP. Frontend is the existing single-file HTML prototype with real API calls. DeepSeek LM (translation) and Qwen VLM (paratext OCR) are already integrated in the pipeline — the GUI only triggers them via mode selection.

**Tech Stack:** FastAPI + Uvicorn (backend), vanilla JS single-file HTML (frontend), Pillow (already in deps), pytest + httpx (tests). No database, no build tools.

```python
"""AMTA GUI — local web server wrapping the translation pipeline.

Endpoints:
  GET  /api/health              — liveness check
  POST /api/translate           — submit a translation job
  GET  /api/jobs/{job_id}       — poll job status

    status: str
    version: str


@app.get("/api/health", response_model=HealthResponse)
def health() -> HealthResponse:
    return HealthResponse(status="ok", version=app.version)
```

```python
class TranslateRequest(BaseModel):
    src_dir: str
    work_id: str
    mode: str = "default"  # "default" or "vlm"
    pages: list[int] | None = None  # None = all pages in src_dir


class JobResponse(BaseModel):
    job_id: str
    status: str  # queued / running / done / failed
    mode: str
    work_id: str
    error: str | None = None


def _run_pipeline_background(job_id: str, src_dir: str, work_id: str,
                             mode: str, pages: list[int] | None) -> None:
    """Run the translation pipeline in a background thread.

    This wraps scripts/run_pipeline.py's core logic without subprocess overhead.
    Default mode: detect,ocr,translate,inpaint,typeset (hayai OCR).
    VLM mode: same stages but vlm_pages = all requested pages.
    """
    import sys
    sys.path.insert(0, str(Path(__file__).resolve().parent.parent.parent.parent / "src"))

    from amta.orchestrator import PipelineConfig, run_pipeline

    _jobs[job_id]["status"] = "running"

    try:
        vlm_pages = pages if mode == "vlm" else []
        config = PipelineConfig(
            work_id=work_id,
            src_dir=Path(src_dir),
            pages=pages,
            vlm_pages=vlm_pages,
            stages=["detect", "ocr", "translate", "inpaint", "typeset"],
            stage_configs={},
            force_rerun=False,
            continue_on_error=True,
        )
        result = run_pipeline(config)
        _jobs[job_id]["status"] = "done" if not result.failed_pages else "failed"
        if result.failed_pages:
            _jobs[job_id]["error"] = f"{len(result.failed_pages)} page(s) failed"
    except Exception as e:
        _jobs[job_id]["status"] = "failed"
        _jobs[job_id]["error"] = str(e)


@app.post("/api/translate", response_model=JobResponse, status_code=202)
def submit_translate(req: TranslateRequest) -> JobResponse:
    if req.mode not in ("default", "vlm"):
        raise HTTPException(status_code=400, detail="mode must be 'default' or 'vlm'")

    job_id = str(uuid.uuid4())[:8]
    _jobs[job_id] = {
        "job_id": job_id,
        "status": "queued",
        "mode": req.mode,
        "work_id": req.work_id,
        "error": None,
    }

    import threading
    t = threading.Thread(
        target=_run_pipeline_background,
        args=(job_id, req.src_dir, req.work_id, req.mode, req.pages),
        daemon=True,
    )
    t.start()

    return JobResponse(**_jobs[job_id])
```

```python
from amta.common.paths import ROOT as AMTA_ROOT


def _workspace_dir(work_id: str) -> Path:
    return AMTA_ROOT / "workspace" / work_id


@app.get("/api/artifacts/{work_id}")
def list_artifacts(work_id: str) -> dict:
    """List final page images for a work_id."""
    ws = _workspace_dir(work_id)
    artifacts = []
    if ws.exists():
        # Final images live in workspace/<work_id>/final/ or similar
        for pattern in ("final", "typeset", "output"):
            d = ws / pattern
            if d.exists():
                for f in sorted(d.glob("*")):
                    if f.suffix.lower() in (".png", ".jpg", ".jpeg", ".webp"):
                        artifacts.append({
                            "page": f.stem,
                            "path": str(f),
                            "url": f"/api/image/{work_id}/{f.stem}",
                        })
                if artifacts:
                    break
    return {"work_id": work_id, "artifacts": artifacts}


@app.get("/api/image/{work_id}/{page}")
def get_image(work_id: str, page: str):
    """Serve a final page image by work_id + page name."""
    ws = _workspace_dir(work_id)
    for pattern in ("final", "typeset", "output"):
        d = ws / pattern
        if d.exists():
            for ext in (".png", ".jpg", ".jpeg", ".webp"):
                f = d / f"{page}{ext}"
                if f.exists():
                    return FileResponse(f)
    raise HTTPException(status_code=404, detail="image not found")
```

```javascript
function pollJob(jobId, workId) {
  var steps = ['Detecting text regions…', 'OCR…', 'Translating…', 'Inpainting…', 'Typesetting…'];
  var stepIdx = 0;
  loadingStep.textContent = steps[0];

  var interval = setInterval(function() {
    fetch('/api/jobs/' + jobId)
      .then(function(r) { return r.json(); })
      .then(function(job) {
        stepIdx = Math.min(stepIdx + 1, steps.length - 1);
        loadingStep.textContent = steps[stepIdx];

        if (job.status === 'done') {
          clearInterval(interval);
          loadingOverlay.classList.remove('show');
          // Load result image
          fetch('/api/artifacts/' + workId)
            .then(function(r) { return r.json(); })
            .then(function(data) {
              if (data.artifacts.length > 0) {
                outputImg.src = data.artifacts[0].url;
                outputImg.classList.add('show');
                emptyOutput.style.display = 'none';
                outputDot.classList.add('done');

                images[currentIndex].translated = true;
                showToast('Translation complete', 'success');

          loadingOverlay.classList.remove('show');
          translateBtn.disabled = false;
```

```python
import shutil
from fastapi import UploadFile, File, Form


@app.post("/api/upload")
async def upload_images(
    files: list[UploadFile] = File(...),
    work_id: str = Form(...),
) -> dict:
    """Accept dropped images, save to a temp src_dir under workspace/<work_id>/input/."""
    ws = _workspace_dir(work_id)
    input_dir = ws / "input"
    if input_dir.exists():
        shutil.rmtree(input_dir)
    input_dir.mkdir(parents=True, exist_ok=True)

    saved = []
    for i, f in enumerate(files, start=1):
        ext = Path(f.filename).suffix if f.filename else ".jpg"
        dest = input_dir / f"{i}{ext}"
        content = await f.read()
        dest.write_bytes(content)
        saved.append(str(dest))

    return {
        "work_id": work_id,
        "src_dir": str(input_dir),
        "page_count": len(saved),
        "pages": list(range(1, len(saved) + 1)),
    }
```

```python
@app.get("/api/export/{work_id}/{page}/png")
def export_png(work_id: str, page: str):
    """Download a final page as PNG."""
    ws = _workspace_dir(work_id)
    for pattern in ("final", "typeset", "output"):
        d = ws / pattern
        if d.exists():
            for ext in (".png", ".jpg", ".jpeg", ".webp"):
                f = d / f"{page}{ext}"
                if f.exists():
                    return FileResponse(
                        f,
                        media_type="image/png",
                        filename=f"{work_id}_{page}.png",
                    )
    raise HTTPException(status_code=404, detail="image not found")


@app.get("/api/export/{work_id}/{page}/psd")
def export_psd(work_id: str, page: str):
    """Download a final page as PSD (editable text layer).

    PSD export is already supported by the typeset stage (commit c2ff355).
    PSD files live alongside final images if the stage config enables it.
    """
    ws = _workspace_dir(work_id)
    for pattern in ("final", "typeset", "output", "psd"):
        d = ws / pattern
        if d.exists():
            f = d / f"{page}.psd"
            if f.exists():
                return FileResponse(
                    f,
                    media_type="image/vnd.adobe.photoshop",
                    filename=f"{work_id}_{page}.psd",
                )
    raise HTTPException(status_code=404, detail="PSD not found — run typeset with PSD export enabled")
```

```python
# tests/test_gui_e2e.py
"""End-to-end GUI test: upload image → submit translation → poll → get result.

This test uses the real pipeline but with a tiny 10x10 white image.
Detection/OCR will find no text, translation short-circuits on empty regions,
and typeset produces an unchanged image. This validates the full HTTP flow
without needing actual manga pages or API keys.

If API keys are not configured, the test skips gracefully.
"""
from __future__ import annotations

import os
import tempfile
import time
from io import BytesIO
from pathlib import Path

import pytest
from fastapi.testclient import TestClient
from PIL import Image

from amta.gui.server import app


def _has_api_keys() -> bool:
    """Check if translation API keys are configured."""
    return bool(os.environ.get("CHAT_API_KEY") or (Path(".env").exists()))


@pytest.mark.skipif(not _has_api_keys(), reason="API keys not configured")
def test_e2e_upload_translate_poll_export():
    client = TestClient(app)

    # 1. Upload a test image
    img_bytes = BytesIO()
    Image.new("RGB", (200, 200), "white").save(img_bytes, format="JPEG")
    img_bytes.seek(0)

    work_id = "e2e-test-" + str(int(time.time()))
    resp = client.post(
        "/api/upload",
        files={"files": ("1.jpg", img_bytes, "image/jpeg")},
        data={"work_id": work_id},
    )
    assert resp.status_code == 200
    upload = resp.json()
    assert upload["page_count"] == 1
    src_dir = upload["src_dir"]

    # 2. Submit translation job (default mode)
    resp = client.post("/api/translate", json={
        "work_id": work_id,
        "src_dir": src_dir,
        "mode": "default",
        "pages": [1],
    })
    assert resp.status_code == 202
    job = resp.json()
    job_id = job["job_id"]
    assert job["status"] in ("queued", "running")

    # 3. Poll until done (max 60s)
    deadline = time.time() + 60
    final_status = None
    while time.time() < deadline:
        resp = client.get(f"/api/jobs/{job_id}")
        assert resp.status_code == 200
        final_status = resp.json()["status"]
        if final_status in ("done", "failed"):
            break
        time.sleep(2)

    assert final_status == "done", f"job ended with status: {final_status}"

    # 4. List artifacts
    resp = client.get(f"/api/artifacts/{work_id}")
    assert resp.status_code == 200
    artifacts = resp.json()["artifacts"]
    # Empty white image may produce no text regions, but pipeline should still
    # produce a final image (inpaint + typeset on empty regions = copy through)
    # If no artifacts, that's acceptable for a blank image — verify the endpoint works
    assert isinstance(artifacts, list)

    # 5. Verify health check still works
    resp = client.get("/api/health")
    assert resp.status_code == 200
    assert resp.json()["status"] == "ok"


def test_e2e_vlm_mode_submission():
    """VLM mode submission should be accepted without running (no DASHSCOPE key needed for submission)."""
    client = TestClient(app)

    with tempfile.TemporaryDirectory() as tmpdir:
        src_dir = Path(tmpdir) / "src"
        src_dir.mkdir()
        Image.new("RGB", (100, 100), "white").save(src_dir / "1.jpg")

        resp = client.post("/api/translate", json={
            "work_id": "e2e-vlm-submit",
            "src_dir": str(src_dir),
            "mode": "vlm",
            "pages": [1],
        })
        assert resp.status_code == 202
        assert resp.json()["mode"] == "vlm"


def test_e2e_invalid_mode_rejected():
    client = TestClient(app)
    resp = client.post("/api/translate", json={
        "work_id": "bad",
        "src_dir": "/tmp",
        "mode": "invalid",
        "pages": [1],
    })
    assert resp.status_code == 400
```

```python
# scripts/run_gui.py
"""Launch the AMTA GUI web server.

Usage:
  uv run python scripts/run_gui.py [--port 8000] [--host 127.0.0.1]

Opens the browser automatically after startup.
"""
from __future__ import annotations

import argparse
import threading
import time
import webbrowser

import uvicorn


def main() -> int:
    ap = argparse.ArgumentParser(description="AMTA GUI — local web server")
    ap.add_argument("--host", default="127.0.0.1")
    ap.add_argument("--port", type=int, default=8000)
    ap.add_argument("--no-browser", action="store_true", help="don't open browser")
    args = ap.parse_args()

    url = f"http://{args.host}:{args.port}"

    if not args.no_browser:
        def _open():
            time.sleep(1.5)
            webbrowser.open(url)
        threading.Thread(target=_open, daemon=True).start()

    print(f"[AMTA GUI] starting on {url}")
    print(f"[AMTA GUI] press Ctrl+C to stop")
    uvicorn.run("amta.gui.server:app", host=args.host, port=args.port, reload=False)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
